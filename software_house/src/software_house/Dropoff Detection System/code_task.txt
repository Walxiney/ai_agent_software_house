# backend/main.py
from fastapi import FastAPI, UploadFile, File
from pydantic import BaseModel
import base64
import os
from dotenv import load_dotenv
from openai import OpenAI

load_dotenv()
openai_client = OpenAI(api_key=os.getenv("OPENAI_KEY"))

app = FastAPI()

class ValidationRequest(BaseModel):
    description: str
    image: str

@app.post("/validate/")
async def validate_dropoff(request: ValidationRequest):
    image_data = request.image
    text_description = request.description

    prompt = [
        {
            "role": "system",
            "content": "You will receive an image as input and a description about a parcel delivered place. Your task is to check if the place where the parcel is in the image match with the description gave by the deliver. You must carefully analyze the image and the description to determine if the parcel is in the correct place."
        },
        {
            "role": "user",
            "content": [
                {
                    "type": "text",
                    "text": text_description,
                },
                {
                    "type": "image_url",
                    "image_url": {
                        "url": f"data:image/png;base64,{image_data}"
                    }
                }
            ]
        }
    ]

    # Call OpenAI API
    response = openai_client.chat.completions.create(
        model="gpt-4o",
        messages=prompt,
        response_format={"type": "json_object"}
    )

    return response['choices'][0]['message']

# frontend/src/App.js
import React, { useState } from 'react';
import './App.css';

function App() {
  const [description, setDescription] = useState('');
  const [image, setImage] = useState(null);
  const [imagePreview, setImagePreview] = useState(null);
  const [result, setResult] = useState('');
  const [reasoning, setReasoning] = useState('');
  const [showReasoning, setShowReasoning] = useState(false);

  const handleImageUpload = (event) => {
    const file = event.target.files[0];
    if (file) {
      const reader = new FileReader();
      reader.onloadend = () => {
        setImage(reader.result.split(',')[1]); // Save base64 image data
        setImagePreview(reader.result);
      };
      reader.readAsDataURL(file);
    }
  };

  const validateDropoff = async () => {
    const response = await fetch('http://localhost:8000/validate/', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({ description, image }),
    });
    
    const data = await response.json();
    setResult(data.answer);
    setReasoning(data.reasoning);
  };

  return (
    <div className="App">
      <h1>Dropoff Detection System</h1>
      <textarea 
        placeholder="Enter description of dropoff location"
        value={description}
        onChange={(e) => setDescription(e.target.value)}
      ></textarea>
      <input type="file" accept=".png, .jpeg, .jpg, .webp" onChange={handleImageUpload} />
      {imagePreview && (
        <div>
          <img src={imagePreview} alt="Preview" onClick={() => setImagePreview(null)} />
        </div>
      )}
      <button onClick={validateDropoff}>Validate</button>
      {result && (
        <div>
          <h2>{result}</h2>
          <button onClick={() => setShowReasoning(!showReasoning)}>
            {showReasoning ? 'Hide Reasoning' : 'Show Reasoning'}
          </button>
          {showReasoning && <p>{reasoning}</p>}
        </div>
      )}
    </div>
  );
}

export default App;

### Installation and Set-up Guide:

1. **Backend Setup:**
   - Install dependencies:
   ```bash
   pip install fastapi uvicorn openai python-dotenv
   ```
   - Run the backend:
   ```bash
   uvicorn main:app --reload
   ```

2. **Frontend Setup:**
   - Create React app:
   ```bash
   npx create-react-app frontend
   cd frontend
   ```
   - Replace the `src/App.js` and `src/App.css` with the provided code.
   - Start the frontend:
   ```bash
   npm start
   ```

### API Documentation:
- **POST `/validate/`**
  - **Request Body:**
    ```json
    {
      "description": "description text here",
      "image": "base64 image string here"
    }
    ```
  - **Response:**
    ```json
    {
      "answer": "Valid" or "Invalid",
      "reasoning": "The reasoning you used to determine the answer."
    }
    ```
